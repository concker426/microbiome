% ── Table 2: Ablation Study ──
\begin{table}[ht]
\centering
\caption{\textbf{Ablation study: contribution of each architectural component.}
All variants use the identical MLP architecture and training procedure.
$\Delta$ shows improvement over variant A (Raw$\rightarrow$MLP, no embedding).}
\label{tab:ablation}
\begin{tabular}{lccccc}
\toprule
\textbf{Variant} & \textbf{ACC} & \textbf{AUC} & \textbf{Sens.} & \textbf{Spec.} & \textbf{$\Delta$ACC} \\
\midrule
A: Raw abundance $\rightarrow$ MLP & 0.8982 & 0.9605 & 0.8817 & 0.9189 & -- \\
B: RandomEmb $\rightarrow$ MLP & 0.9162 & 0.9762 & 0.9140 & 0.9189 & +1.80\% \\
C: SimpleEmb $\rightarrow$ Linear & 0.4431 & 0.3362 & 0.0645 & 0.9189 & $-$45.51\% \\
D: \textbf{SimpleEmb $\rightarrow$ MLP (ProCyon v2)} & \textbf{0.9257} & \textbf{0.9750} & \textbf{0.9161} & \textbf{0.9378} & \textbf{+2.75\%} \\
\bottomrule
\end{tabular}

\vspace{4pt}
\textit{Interpretation:} A$\rightarrow$B (+1.80\%): random projection from 86-dim sparse to 768-dim dense provides modest benefit.
B$\rightarrow$D (+0.95\%): training the embedding end-to-end adds further gain.
C (44.31\%): the embedding space is not linearly separable---a non-linear MLP is essential.
\end{table}
